## 📦 优化组件使用

### 1. OptimizedList - 优化列表组件

#### 基础用法
```typescript
import OptimizedList from '@/components/Common/OptimizedList';

<OptimizedList
  data={projects}
  renderItem={(project) => (
    <Card>
      <h3>{project.name}</h3>
      <p>{project.description}</p>
    </Card>
  )}
/>
```

#### 启用虚拟滚动（大列表必备）
```typescript
<OptimizedList
  data={largeDataArray}  // 1000+ 项
  renderItem={(item) => <ItemCard item={item} />}
  virtualScroll          // 启用虚拟滚动
  itemHeight={80}        // 每项高度
  pageSize={20}          // 每页数量
/>
```

#### 启用搜索功能
```typescript
<OptimizedList
  data={projects}
  renderItem={(project) => <ProjectCard project={project} />}
  searchable             // 启用搜索
  searchKeys={['name', 'description', 'id']}  // 搜索字段
/>
```

#### 无限滚动加载
```typescript
const [hasMore, setHasMore] = useState(true);

<OptimizedList
  data={items}
  renderItem={(item) => <Item item={item} />}
  onLoadMore={async () => {
    const newItems = await fetchMoreItems();
    setItems([...items, ...newItems]);
    setHasMore(newItems.length > 0);
  }}
  hasMore={hasMore}
  loading={loading}
/>
```

### 2. useOptimizedQuery - 智能查询Hook

#### 基础用法
```typescript
import useOptimizedQuery from '@/hooks/useOptimizedQuery';

function ProjectList() {
  const { data, loading, error, refetch } = useOptimizedQuery(
    () => projectApi.getAll()
  );

  if (loading) return <Spin />;
  if (error) return <Alert message={error.message} />;
  
  return <List dataSource={data} />;
}
```

#### 启用缓存
```typescript
const { data, loading } = useOptimizedQuery(
  () => projectApi.getAll(),
  {
    cacheKey: 'projects',           // 缓存键
    cacheTTL: 5 * 60 * 1000,       // 5分钟过期
  }
);

// 第二次调用会直接从缓存读取，无需请求
```

#### 自动重试
```typescript
const { data, loading, error } = useOptimizedQuery(
  () => projectApi.getAll(),
  {
    retryCount: 3,                  // 失败重试3次
    retryDelay: 1000,               // 每次延迟1秒
  }
);
```

#### 防抖查询
```typescript
const { data, loading } = useOptimizedQuery(
  () => searchApi.search(keyword),
  {
    debounceMs: 300,                // 300ms防抖
    enabled: keyword.length > 0,    // 条件启用
  }
);
```

#### 成功/失败回调
```typescript
const { data, loading } = useOptimizedQuery(
  () => projectApi.create(newProject),
  {
    onSuccess: (data) => {
      message.success('创建成功');
      navigate(`/project/${data.id}`);
    },
    onError: (error) => {
      message.error(`创建失败: ${error.message}`);
    }
  }
);
```

#### 手动刷新
```typescript
const { data, refetch, reset } = useOptimizedQuery(
  () => projectApi.getAll(),
  { cacheKey: 'projects' }
);

// 刷新数据（忽略缓存）
<Button onClick={refetch}>刷新</Button>

// 重置状态（清除缓存）
<Button onClick={reset}>重置</Button>
```

### 4. 缓存管理

#### 内存缓存
```typescript
import { MemoryCache } from '@/utils/optimizedPerformance';

// 创建缓存实例
const cache = new MemoryCache({
  ttl: 5 * 60 * 1000,    // 5分钟过期
  maxSize: 100           // 最多100项
});

// 设置缓存
cache.set('user-123', userData);

// 获取缓存
const user = cache.get('user-123');

// 检查是否存在
if (cache.has('user-123')) {
  // ...
}

// 删除缓存
cache.delete('user-123');

// 清空所有缓存
cache.clear();

// 清理过期项
const cleaned = cache.cleanup();
```

### 5. 批处理器

#### 批量API请求
```typescript
import { BatchProcessor } from '@/utils/optimizedPerformance';

// 创建批处理器
const batcher = new BatchProcessor(
  async (items) => {
    // 批量处理
    await api.batchUpdate(items);
  },
  {
    maxBatchSize: 50,      // 最多50项一批
    maxWaitTime: 1000      // 最多等待1秒
  }
);

// 添加项（自动批处理）

// 手动触发

```

import React from 'react';

function ProjectListPage() {

### 示例2：优化搜索功能

```typescript
import React, { useState } from 'react';
import { Input } from 'antd';
import useOptimizedQuery from '@/hooks/useOptimizedQuery';
import { debounce } from '@/utils/optimizedPerformance';

function SearchPage() {
  const [keyword, setKeyword] = useState('');

  const { data: results, loading } = useOptimizedQuery(
    () => searchApi.search(keyword),
    {
      cacheKey: `search-${keyword}`,
      debounceMs: 300,
      enabled: keyword.length > 2
    }
  );

  const handleSearch = debounce((value) => {
    setKeyword(value);
  }, 300);

  return (
    <>
      <Input
        placeholder="搜索..."
        onChange={(e) => handleSearch(e.target.value)}
      />
      <OptimizedList
        data={results || []}
        loading={loading}
        renderItem={(item) => <SearchResult item={item} />}
      />
    </>
  );
}
```

## 🔧 故障排查

### 问题：虚拟滚动不工作
**解决方案：**
1. 确保设置了 `itemHeight`
2. 检查容器是否有固定高度
3. 验证数据格式正确

### 问题：缓存不生效
**解决方案：**
1. 确保设置了 `cacheKey`
2. 检查 `cacheTTL` 是否过短
3. 验证缓存未被清除

### 问题：性能监控无数据
**解决方案：**
1. 确保浏览器支持 Performance API
2. 检查是否正确调用 `mark` 和 `measure`
3. 验证标记名称匹配
