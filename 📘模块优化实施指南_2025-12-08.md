```typescript
    } catch (error) {
      message.error('加载失败');

    }

  fetchProjects();

import useOptimizedQuery from '../hooks/useOptimizedQuery';

const { data: projects, loading, error, refetch } = useOptimizedQuery(

  {
    cacheKey: 'projects',
    cacheTTL: 5 * 60 * 1000,
    retryCount: 3,
    onError: (error) => message.error('加载失败')
  }
);
```

```typescript
// client/src/pages/TaskManagement.tsx
const [selectedTasks, setSelectedTasks] = useState<string[]>([]);

const handleBatchOperation = async (operation: string) => {
  switch (operation) {
    case 'assign':
      // 批量分配
      await taskApi.batchAssign(selectedTasks, assignee);
      break;
    case 'status':
      // 批量修改状态
      await taskApi.batchUpdateStatus(selectedTasks, newStatus);
      break;
    case 'delete':
      // 批量删除
      await taskApi.batchDelete(selectedTasks);
      break;
  }
  message.success(`已${operation} ${selectedTasks.length} 个任务`);
  setSelectedTasks([]);
  refetch();
};

<Space>
  <Button
    disabled={selectedTasks.length === 0}
    onClick={() => handleBatchOperation('assign')}
  >
    批量分配
  </Button>
  <Button
    disabled={selectedTasks.length === 0}
    onClick={() => handleBatchOperation('status')}
  >
    批量修改状态
  </Button>
  <Button
    danger
    disabled={selectedTasks.length === 0}
    onClick={() => handleBatchOperation('delete')}
  >
    批量删除
  </Button>
</Space>
```

#### 3. 搜索过滤功能

**高级搜索组件**

```typescript
// client/src/components/Common/AdvancedSearch.tsx
import React, { useState } from 'react';
import { Form, Input, Select, DatePicker, Button, Space } from 'antd';
import { SearchOutlined, FilterOutlined, ReloadOutlined } from '@ant-design/icons';

interface AdvancedSearchProps {
  fields: Array<{
    name: string;
    label: string;
    type: 'text' | 'select' | 'date' | 'dateRange';
    options?: Array<{ label: string; value: any }>;
  }>;
  onSearch: (values: any) => void;
  onReset: () => void;
}

const AdvancedSearch: React.FC<AdvancedSearchProps> = ({ fields, onSearch, onReset }) => {
  const [form] = Form.useForm();
  const [expanded, setExpanded] = useState(false);

  const handleSearch = () => {
    const values = form.getFieldsValue();
    onSearch(values);
  };

  const handleReset = () => {
    form.resetFields();
    onReset();
  };

  return (
    <Form form={form} layout="inline">
      {fields.slice(0, expanded ? fields.length : 3).map((field) => (
        <Form.Item key={field.name} name={field.name} label={field.label}>
          {field.type === 'text' && <Input placeholder={`请输入${field.label}`} />}
          {field.type === 'select' && (
            <Select placeholder={`请选择${field.label}`} options={field.options} />
          )}
          {field.type === 'date' && <DatePicker />}
          {field.type === 'dateRange' && <DatePicker.RangePicker />}
        </Form.Item>
      ))}
      <Form.Item>
        <Space>
          <Button type="primary" icon={<SearchOutlined />} onClick={handleSearch}>
            搜索
          </Button>
          <Button icon={<ReloadOutlined />} onClick={handleReset}>
            重置
          </Button>
          {fields.length > 3 && (
            <Button
              type="link"
              icon={<FilterOutlined />}
              onClick={() => setExpanded(!expanded)}
            >
              {expanded ? '收起' : '展开'}
            </Button>
          )}
        </Space>
      </Form.Item>
    </Form>
  );
};

export default AdvancedSearch;
```
